class EmployeeReader {
    public void display(String fileName) throws IOException {
        BufferedReader reader = new BufferedReader(new FileReader(fileName));
        StringBuilder output = new StringBuilder();
        String id;

        while ((id = reader.readLine()) != null) {
            String name = reader.readLine();
            String designation = reader.readLine();
            String salary = reader.readLine();

            if (output.length() > 0) {
                output.append("\n");
            }
            output.append("Employee ID: ").append(id).append("\n");
            output.append("Employee Name: ").append(name).append("\n");
            output.append("Employee Designation: ").append(designation).append("\n");
            output.append("Employee Salary: ").append(salary);
        }
        reader.close();

        System.out.print(output.toString());
    }
}

class SalaryReader {
    public void readAndDisplay(String sourceFile, String destinationFile) {
        try {
            BufferedReader reader = new BufferedReader(new FileReader(sourceFile));
            BufferedWriter writer = new BufferedWriter(new FileWriter(destinationFile));
            
            String line;
            while ((line = reader.readLine()) != null) {
                writer.write(line);
                writer.newLine();
            }
            reader.close();
            writer.close();

            BufferedReader destReader = new BufferedReader(new FileReader(destinationFile));
            StringBuilder output = new StringBuilder();
            while ((line = destReader.readLine()) != null) {
                if (output.length() > 0) {
                    output.append("\n");
                }
                output.append(line);
            }
            destReader.close();

            System.out.print(output.toString());
        } catch (IOException e) {
            System.out.println("An error occurred: " + e.getMessage());
        }
    }
}

class TextAnalyzer {
    public void analyzeFile(String fileName) throws IOException {
        BufferedReader reader = new BufferedReader(new FileReader(fileName));
        int charCount = 0;
        int lineCount = 0;
        int wordCount = 0;
        String line;

        while ((line = reader.readLine()) != null) {
            lineCount++;
            charCount += line.length();

            String trimmed = line.trim();
            if (!trimmed.isEmpty()) {
                String[] words = trimmed.split("\\s+");
                wordCount += words.length;
            }
        }
        reader.close();

        System.out.println("No of characters: " + charCount);
        System.out.println("No of lines: " + lineCount);
        System.out.print("No of words: " + wordCount);
    }
}

class BinaryFileComparator {

        FileInputStream fis1 = new FileInputStream(file1);
        int decimal1 = fis1.read();
        fis1.close();

        FileInputStream fis2 = new FileInputStream(file2);
        int decimal2 = fis2.read();
        fis2.close();

        System.out.println("Content of " + file1 + ": " + binary1 + " (Decimal: " + decimal1 + ")");
        System.out.println("Content of " + file2 + ": " + binary2 + " (Decimal: " + decimal2 + ")");

        if (decimal1 == decimal2) {
            System.out.println("Two files are equal");
        } else {
            System.out.println("Two files are not equal");
        }
    }
}

class SentimentAnalyzer 
{
    String getUserInput() 
    {
        Scanner sc = new Scanner(System.in);
        String input = "";
        if (sc.hasNextLine()) {
            input = sc.nextLine();
        }
        sc.close();
        return input;
    }

    void writeDataToFile(String data, String fileName) throws IOException 
    {
        FileOutputStream fos = new FileOutputStream(fileName);
        DataOutputStream dos = new DataOutputStream(fos);
        dos.writeUTF(data);
        dos.close();
        fos.close();
    }

    String analyzeSentiment(String fileName) throws IOException 
    {
        FileInputStream fis = new FileInputStream(fileName);
        DataInputStream dis = new DataInputStream(fis);
        String sentence = dis.readUTF();
        dis.close();
        fis.close();

        String lowerSentence = sentence.toLowerCase();
        String[] positiveKeywords = {"happy", "good", "excellent", "positive"};
        String[] negativeKeywords = {"sad", "bad", "terrible", "negative"};

        for (String word : positiveKeywords) {
            if (lowerSentence.contains(word)) {
                return "Positive";
            }
        }

        for (String word : negativeKeywords) {
            if (lowerSentence.contains(word)) {
                return "Negative";
            }
        }

        return "Neutral";
    }

    void displayResult(String fileName) throws IOException 
    {
        FileInputStream fis = new FileInputStream(fileName);
        DataInputStream dis = new DataInputStream(fis);
        String result = dis.readUTF();
        dis.close();
        fis.close();

        System.out.print(result);
    }
}
