import java.util.*;

class EmailAnalyzer {
    public void analyzeDomains(List<String> emails) {
        TreeMap<String, Integer> domainCounts = new TreeMap<>();

        for (String email : emails) {
            int atIndex = email.indexOf('@');
            if (atIndex != -1 && atIndex + 1 < email.length()) {
                String domain = email.substring(atIndex + 1).trim();
                domainCounts.put(domain, domainCounts.getOrDefault(domain, 0) + 1);
            }
        }

        StringBuilder sb = new StringBuilder();
        sb.append("Domain Frequency:");
        for (Map.Entry<String, Integer> entry : domainCounts.entrySet()) {
            sb.append("\n").append(entry.getKey()).append(": ").append(entry.getValue());
        }
        System.out.print(sb.toString());
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) {
            sc.close();
            return;
        }
        int n = Integer.parseInt(sc.nextLine().trim());
        List<String> emails = new ArrayList<>();
        for (int i = 0; i < n; i++) {
            if (sc.hasNextLine()) {
                emails.add(sc.nextLine().trim());
            }
        }
        sc.close();

        EmailAnalyzer analyzer = new EmailAnalyzer();
        analyzer.analyzeDomains(emails);
    }
}

class TwoSum {
    public static int[] findTwoSum(int[] arr, int target) {
        HashMap<Long, Integer> seen = new HashMap<>();
        for (int i = 0; i < arr.length; i++) {
            long complement = (long) target - arr[i];
            if (seen.containsKey(complement)) {
                return new int[]{seen.get(complement), i};
            }
            seen.put((long) arr[i], i);
        }
        return new int[]{-1, -1};
    }
}

class ProductAnalyzer {
    public void analyzeProductRatings(List<String> lines) {
        TreeMap<Integer, Double> totalRatings = new TreeMap<>();
        TreeMap<Integer, Integer> ratingCounts = new TreeMap<>();

        for (String line : lines) {
            String[] parts = line.trim().split("\\s+");
            if (parts.length == 2) {
                int productId = Integer.parseInt(parts[0]);
                double rating = Double.parseDouble(parts[1]);

                totalRatings.put(productId, totalRatings.getOrDefault(productId, 0.0) + rating);
                ratingCounts.put(productId, ratingCounts.getOrDefault(productId, 0) + 1);
            }
        }

        TreeMap<Integer, Double> avgRatings = new TreeMap<>();
        for (int productId : totalRatings.keySet()) {
            double avg = totalRatings.get(productId) / ratingCounts.get(productId);
            avgRatings.put(productId, avg);
        }

        StringBuilder sb = new StringBuilder();
        sb.append("Product ID and Average Rating:");
        for (Map.Entry<Integer, Double> entry : avgRatings.entrySet()) {
            sb.append(String.format("\n%d: %.2f", entry.getKey(), entry.getValue()));
        }
        System.out.print(sb.toString());
    }
}

class WordAnalyzer {
    public void analyzeWordFrequency(List<String> lines) {
        TreeMap<String, Integer> freqMap = new TreeMap<>(String.CASE_INSENSITIVE_ORDER);
        Map<String, String> originalCase = new HashMap<>();

        for (String line : lines) {
            String[] words = line.split("[^a-zA-Z0-9]+");
            for (String word : words) {
                if (word.isEmpty()) {
                    continue;
                }
                String lower = word.toLowerCase();
                if (!originalCase.containsKey(lower)) {
                    originalCase.put(lower, word);
                }
                freqMap.put(word, freqMap.getOrDefault(word, 0) + 1);
            }
        }

        StringBuilder sb = new StringBuilder();
        sb.append("Word Frequency:");
        for (Map.Entry<String, Integer> entry : freqMap.entrySet()) {
            String displayWord = originalCase.get(entry.getKey().toLowerCase());
            sb.append("\n").append(displayWord).append(": ").append(entry.getValue());
        }
        System.out.print(sb.toString());
    }
}

import java.util.*;

class Student implements Comparable<Student> {
    int rollNo;
    String name;
    int attendance;

    Student(int rollNo, String name) {
        this.rollNo = rollNo;
        this.name = name;
        this.attendance = 0;
    }

    @Override
    public int compareTo(Student other) {
        return Integer.compare(this.rollNo, other.rollNo);
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }
        int n = sc.nextInt();

        TreeSet<Student> students = new TreeSet<>();
        StringBuilder output = new StringBuilder();

        for (int i = 0; i < n; i++) {
            String cmd = sc.next();
            if (cmd.equals("A")) {
                int rollNo = sc.nextInt();
                String name = sc.next();
                students.add(new Student(rollNo, name));
            } else if (cmd.equals("M")) {
                int rollNo = sc.nextInt();
                for (Student s : students) {
                    if (s.rollNo == rollNo) {
                        s.attendance++;
                        break;
                    }
                }
            } else if (cmd.equals("D")) {
                for (Student s : students) {
                    if (output.length() > 0) {
                        output.append("\n");
                    }
                    output.append(s.rollNo).append(" ").append(s.name).append(" ").append(s.attendance);
                }
            }
        }
        sc.close();

        System.out.print(output.toString());
    }
}
