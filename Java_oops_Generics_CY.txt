import java.math.BigInteger;
import java.util.Scanner;
import java.util.function.IntFunction;

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }
        int n = sc.nextInt();
        sc.close();

        IntFunction<BigInteger> factorial = num -> {
            BigInteger result = BigInteger.ONE;
            for (int i = 2; i <= num; i++) {
                result = result.multiply(BigInteger.valueOf(i));
            }
            return result;
        };

        System.out.print(factorial.apply(n));
    }
}

import java.util.Scanner;

@FunctionalInterface
interface WordCategorizer {
    String categorize(String word);
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextLine()) {
            sc.close();
            return;
        }
        String sentence = sc.nextLine().trim();
        sc.close();

        if (sentence.isEmpty()) {
            return;
        }

        WordCategorizer checkLength = word -> word.length() <= 3 ? "Short" : "Long";

        String[] words = sentence.split("\\s+");
        StringBuilder sb = new StringBuilder();
        for (int i = 0; i < words.length; i++) {
            if (i > 0) {
                sb.append(" ");
            }
            sb.append(checkLength.categorize(words[i]));
        }

        System.out.print(sb.toString());
    }
}

import java.util.Scanner;

public class Main {
    public static <T> void reverseArray(T[] array) {
        int left = 0;
        int right = array.length - 1;
        while (left < right) {
            T temp = array[left];
            array[left] = array[right];
            array[right] = temp;
            left++;
            right--;
        }
    }

    public static <T> String arrayToString(T[] array) {
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < array.length; i++) {
            sb.append(array[i]);
            if (i < array.length - 1) {
                sb.append(", ");
            }
        }
        sb.append("]");
        return sb.toString();
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }
        int size = sc.nextInt();

        Integer[] arr = new Integer[size];
        for (int i = 0; i < size; i++) {
            arr[i] = sc.nextInt();
        }
        sc.close();

        System.out.println(arrayToString(arr));
        reverseArray(arr);
        System.out.print(arrayToString(arr));
    }
}

import java.util.Arrays;
import java.util.Scanner;

class Swapper<T> {
    private T[] array;

    public Swapper(T[] array) {
        this.array = array;
    }

    public boolean swap(int a, int b) {
        if (a < 0 || a >= array.length || b < 0 || b >= array.length) {
            return false;
        }
        T temp = array[a];
        array[a] = array[b];
        array[b] = temp;
        return true;
    }

    public T[] getArray() {
        return array;
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (!sc.hasNextInt()) {
            sc.close();
            return;
        }
        int n = sc.nextInt();

        Integer[] arr = new Integer[n];
        for (int i = 0; i < n; i++) {
            arr[i] = sc.nextInt();
        }
        int a = sc.nextInt();
        int b = sc.nextInt();
        sc.close();

        System.out.println("Original Array: " + Arrays.toString(arr));

        Swapper<Integer> swapper = new Swapper<>(arr);
        if (swapper.swap(a, b)) {
            System.out.print("Array after swapping elements at indices " + a + " and " + b + ": " + Arrays.toString(swapper.getArray()));
        } else {
            System.out.print("Invalid indices. Swap operation aborted.");
        }
    }
}

import java.util.Scanner;

class Rectangle<T extends Number> {
    private T length;
    private T width;

    public Rectangle(T length, T width) {
        this.length = length;
        this.width = width;
    }

    public double calculateArea() {
        return length.doubleValue() * width.doubleValue();
    }
}

public class Main {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        double length = sc.nextDouble();
        double width = sc.nextDouble();
        sc.close();

        Rectangle<Double> rect = new Rectangle<>(length, width);
        System.out.printf("Area of the rectangle: %.2f", rect.calculateArea());
    }
}
